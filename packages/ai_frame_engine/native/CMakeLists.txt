cmake_minimum_required(VERSION 3.22)
project(oplayer_rife LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(NCNN_VULKAN ON CACHE BOOL "" FORCE)
set(NCNN_SIMPLEVK ON CACHE BOOL "" FORCE)
set(NCNN_SHARED_LIB OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_BENCHMARK OFF CACHE BOOL "" FORCE)
set(NCNN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(NCNN_INT8 OFF CACHE BOOL "" FORCE)
set(NCNN_OPENMP OFF CACHE BOOL "" FORCE)
set(NCNN_DISABLE_EXCEPTION OFF CACHE BOOL "" FORCE)
set(NCNN_INSTALL_SDK OFF CACHE BOOL "" FORCE)
set(NCNN_AVX512 OFF CACHE BOOL "" FORCE)
set(NCNN_AVXVNNI OFF CACHE BOOL "" FORCE)
set(NCNN_AVXVNNIINT8 OFF CACHE BOOL "" FORCE)
set(NCNN_AVXVNNIINT16 OFF CACHE BOOL "" FORCE)
set(NCNN_AVXNECONVERT OFF CACHE BOOL "" FORCE)
set(NCNN_AVX512VNNI OFF CACHE BOOL "" FORCE)
set(NCNN_AVX512BF16 OFF CACHE BOOL "" FORCE)
set(NCNN_AVX512FP16 OFF CACHE BOOL "" FORCE)
if(MSVC)
  add_compile_options(/MP4)
endif()
include(FetchContent)
FetchContent_Declare(ncnn
  URL https://github.com/Tencent/ncnn/releases/download/20261020/ncnn-20261020-full-source.zip
  URL_HASH SHA256=699d9acc67489eee0993a551ca1ebc247bbf0f4e33babe1dec810e6a9a52b988
  SOURCE_SUBDIR oplayer-populate-only)
FetchContent_MakeAvailable(ncnn)
# Only the operators needed by the pinned RIFE network and ncnn pixel helpers.
set(required_layers BinaryOp Concat Convolution Crop Deconvolution Eltwise Input Interp
  PixelShuffle ReLU Sigmoid Split Packing Cast Padding Gemm InnerProduct Flatten Reshape)
file(STRINGS "${ncnn_SOURCE_DIR}/src/CMakeLists.txt" layer_lines REGEX "^ncnn_add_layer\\(")
foreach(line IN LISTS layer_lines)
  string(REGEX REPLACE "^ncnn_add_layer\\(([A-Za-z0-9_]+).*" "\\1" layer "${line}")
  string(TOLOWER "${layer}" key)
  if(layer IN_LIST required_layers)
    set(WITH_LAYER_${key} ON CACHE BOOL "" FORCE)
  else()
    set(WITH_LAYER_${key} OFF CACHE BOOL "" FORCE)
  endif()
endforeach()
add_subdirectory("${ncnn_SOURCE_DIR}" "${ncnn_BINARY_DIR}")
file(GLOB shaders CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/rife/*.comp")
set(shader_headers)
foreach(shader IN LISTS shaders)
  get_filename_component(name "${shader}" NAME)
  set(header "${CMAKE_CURRENT_BINARY_DIR}/${name}.hex.h")
  add_custom_command(OUTPUT "${header}" COMMAND ${CMAKE_COMMAND}
    -DSHADER_SRC=${shader} -DSHADER_COMP_HEADER=${header}
    -P ${CMAKE_CURRENT_SOURCE_DIR}/rife/generate_shader_comp_header.cmake
    DEPENDS "${shader}" rife/generate_shader_comp_header.cmake)
  list(APPEND shader_headers "${header}")
endforeach()
add_library(oplayer_rife SHARED engine.cpp npu.cpp npu_android.cpp rife/rife.cpp rife/warp.cpp ${shader_headers})
target_include_directories(oplayer_rife PRIVATE rife "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(oplayer_rife PRIVATE ncnn ${CMAKE_DL_LIBS})
set_target_properties(oplayer_rife PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(MSVC)
  target_link_options(oplayer_rife PRIVATE "/MAP:${CMAKE_CURRENT_BINARY_DIR}/oplayer_rife.map")
  target_link_libraries(oplayer_rife PRIVATE ole32)
else()
  target_compile_options(oplayer_rife PRIVATE -fexceptions)
endif()
if(UNIX)
  find_package(Threads REQUIRED)
  target_link_libraries(oplayer_rife PRIVATE Threads::Threads)
endif()
if(ANDROID)
  target_link_options(oplayer_rife PRIVATE "-Wl,-z,max-page-size=16384")
endif()
if(UNIX AND NOT APPLE)
  target_link_options(oplayer_rife PRIVATE "-Wl,--exclude-libs,ALL")
endif()
